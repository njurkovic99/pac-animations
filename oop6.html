C++ Aggregation & Object Conversion

Understanding class relationships and type conversions

// Aggregation and Object Conversion Demo
#include <iostream>
#include <string>
using namespace std;

class Engine {
private:
    string type;
    int horsepower;
public:
    Engine(string t, int hp) : type(t), horsepower(hp) {
        cout << "Engine created: " << type << endl;
    }
    
    // Conversion operator to int (returns horsepower)
    operator int() const { return horsepower; }
    
    string getType() const { return type; }
};

class Car {
private:
    string model;
    Engine* engine;  // Aggregation - Car HAS-A Engine
    double price;
public:
    Car(string m, double p) : model(m), engine(nullptr), price(p) {
        cout << "Car created: " << model << endl;
    }
    
    void installEngine(Engine* e) {
        engine = e;
        cout << "Engine installed in " << model << endl;
    }
    
    // Conversion operator to double (returns price)
    operator double() const { return price; }
    
    int getHorsepower() const {
        if (engine) {
            return static_cast<int>(*engine);  // Object conversion
        }
        return 0;
    }
};

int main() {
    cout << "=== Starting Program ===" << endl;
    
    // Create an Engine object
    Engine v8Engine("V8 Turbo", 450);
    
    // Create a Car object
    Car sportsCar("Ferrari F8", 280000.0);
    
    // Aggregation: Car uses Engine (doesn't own it)
    sportsCar.installEngine(&v8Engine);
    
    // Object Conversion: Engine to int
    int hp = v8Engine;  // Implicit conversion
    cout << "Engine horsepower (via conversion): " << hp << endl;
    
    // Object Conversion: Car to double
    double carPrice = sportsCar;  // Implicit conversion
    cout << "Car price (via conversion): $" << carPrice << endl;
    
    // Using conversion within member function
    cout << "Total HP from car: " << sportsCar.getHorsepower() << endl;
    
    cout << "=== Program Complete ===" << endl;
    return 0;
}
