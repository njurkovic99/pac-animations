C++ OOP: Static Members, Copy Constructor & Operator Overloading
📁 Counter.h
class Counter {
private:
    int value;
    static int totalCount;  // Static member
    
public:
    // Constructor
    Counter(int v = 0) : value(v) {
        totalCount++;
    }
    
    // Copy constructor
    Counter(const Counter& other) {
        value = other.value;
        totalCount++;
    }
    
    // Operator overloading
    Counter operator+(const Counter& other) {
        return Counter(value + other.value);
    }
    
    void setValue(int v) { value = v; }
    int getValue() const { return value; }
    static int getTotal() { return totalCount; }
};

// Initialize static member
int Counter::totalCount = 0;
📁 main.cpp
#include <iostream>
#include "Counter.h"
using namespace std;

int main() {
    // Create instance
    Counter c1(10);
    cout << "c1 value: " << c1.getValue() << endl;
    cout << "Total counters: " << Counter::getTotal() << endl;
    
    // Member-wise assignment
    Counter c2;
    c2 = c1;  // Assignment
    cout << "c2 value after assignment: " << c2.getValue() << endl;
    
    // Copy constructor
    Counter c3(c1);  // Copy constructor called
    cout << "c3 value: " << c3.getValue() << endl;
    cout << "Total counters: " << Counter::getTotal() << endl;
    
    // Operator overloading
    Counter c4 = c1 + c3;  // Uses operator+
    cout << "c4 value (c1 + c3): " << c4.getValue() << endl;
    cout << "Total counters: " << Counter::getTotal() << endl;
    
    return 0;
}
