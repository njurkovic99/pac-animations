C++ Constructors, Overloading and Private Member Functions

📄 Student.h
class Student {
private:
    string name;
    int id;
    double gpa;
    
    // Private member function
    void validateGPA() {
        if (gpa < 0.0) gpa = 0.0;
        if (gpa > 4.0) gpa = 4.0;
    }

public:
    // Default constructor
    Student() {
        name = "Unknown";
        id = 0;
        gpa = 0.0;
        cout << "Default constructor called" << endl;
    }
    
    // Parameterized constructor (overloaded)
    Student(string n, int i) {
        name = n;
        id = i;
        gpa = 0.0;
        cout << "2-param constructor for " << name << endl;
    }
    
    // Another overloaded constructor
    Student(string n, int i, double g) {
        name = n;
        id = i;
        gpa = g;
        validateGPA();  // Calling private function
        cout << "3-param constructor for " << name << endl;
    }
    
    void display() {
        cout << "Name: " << name << ", ID: " << id 
             << ", GPA: " << gpa << endl;
    }
};
📄 main.cpp
#include <iostream>
#include "Student.h"
using namespace std;

int main() {
    // Using default constructor
    Student s1;
    s1.display();
    
    // Using 2-parameter constructor
    Student s2("Alice", 101);
    s2.display();
    
    // Using 3-parameter constructor
    Student s3("Bob", 102, 3.75);
    s3.display();
    
    // Testing GPA validation (private function)
    Student s4("Charlie", 103, 5.0);
    s4.display();  // GPA should be capped at 4.0
    
    return 0;
}
